<!-- JavaScript Logic -->

<script>

let state = {

vaultCash: 100000,

inventory: [],

transactions: [{ date: new Date().toLocaleDateString('ar-EG'), desc: 'رأس المال الافتتاحي', amount: 100000 }]

};

 

window.onload = function() {

const saved = localStorage.getItem('altaif_simple_state');

if(saved) { try { state = JSON.parse(saved); } catch(e){} }

renderAll();

};

 

function save() {

localStorage.setItem('altaif_simple_state', JSON.stringify(state));

renderAll();

}

 

function switchTab(tabId, btn) {

document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));

document.getElementById('tab-' + tabId).classList.remove('hidden');

document.querySelectorAll('.nav-tab').forEach(b => b.classList.remove('active'));

btn.classList.add('active');

}

 

function handleBuy(e) {

e.preventDefault();

const name = document.getElementById('buyName').value;

const karat = document.getElementById('buyKarat').value;

const weight = parseFloat(document.getElementById('buyWeight').value) || 0;

const cost = parseFloat(document.getElementById('buyCost').value) || 0;

 

if(cost > state.vaultCash) {

alert('عفواً، لا توجد نقدية كافية بالخزينة!');

return;

}

 

state.vaultCash -= cost;

state.inventory.push({

id: 'G-' + Math.floor(1000 + Math.random() * 9000),

name, karat, weight, cost

});

 

state.transactions.unshift({

date: new Date().toLocaleDateString('ar-EG'),

desc: 'شراء: ' + name,

amount: -cost

});

 

document.getElementById('buyName').value = '';

document.getElementById('buyWeight').value = '';

document.getElementById('buyCost').value = '';

save();

alert('تم شراء القطعة وإضافتها للمخزون بنجاح!');

switchTab('inventory', document.querySelectorAll('.nav-tab')[3]);

}

 

function updateSellDetails() {

const id = document.getElementById('sellItemSelect').value;

const item = state.inventory.find(i => i.id === id);

const box = document.getElementById('sellInfoBox');

if(item) {

box.innerHTML = `القطعة: ${item.name} | العيار: ${item.karat} | الوزن: ${item.weight} جم | التكلفة: ${item.cost} ج.م`;

document.getElementById('sellPriceOverride').value = Math.round(item.cost * 1.15);

} else {

box.innerHTML = 'اختر قطعة لعرض تفاصيلها...';

}

}

 

function handleSell(e) {

e.preventDefault();

const id = document.getElementById('sellItemSelect').value;

const item = state.inventory.find(i => i.id === id);

if(!item) return alert('الرجاء اختيار قطعة صحيحة');

 

const sellPrice = parseFloat(document.getElementById('sellPriceOverride').value) || 0;

state.vaultCash += sellPrice;

 

state.inventory = state.inventory.filter(i => i.id !== id);

state.transactions.unshift({

date: new Date().toLocaleDateString('ar-EG'),

desc: 'بيع: ' + item.name,

amount: sellPrice

});

 

save();

alert('تم إتمام البيع وتحديث الخزينة بنجاح!');

switchTab('dashboard', document.querySelectorAll('.nav-tab')[0]);

}

 

function deleteItem(id) {

if(confirm('حذف هذه القطعة من المخزون؟')) {

state.inventory = state.inventory.filter(i => i.id !== id);

save();

}

}

 

function renderAll() {

document.getElementById('dashVault').innerText = Math.round(state.vaultCash).toLocaleString() + ' ج.م';

document.getElementById('dashCount').innerText = state.inventory.length + ' قطعة';

document.getElementById('vaultCashDisplay').innerText = Math.round(state.vaultCash).toLocaleString() + ' ج.م';

 

// Inventory Table

const invBody = document.getElementById('inventoryTableBody');

const sellSelect = document.getElementById('sellItemSelect');

invBody.innerHTML = '';

sellSelect.innerHTML = '<option value="">-- اختر قطعة للبيع --</option>';

 

if(state.inventory.length === 0) {

invBody.innerHTML = '<tr><td colspan="6" style="color: #A0AEC0;">المخزون فارغ حالياً</td></tr>';

} else {

state.inventory.forEach(i => {

invBody.innerHTML += `

<tr>

<td>${i.id}</td>

<td>${i.name}</td>

<td>${i.karat}</td>

<td>${i.weight} جم</td>

<td>${i.cost} ج.م</td>

<td><button onclick="deleteItem('${i.id}')" style="background:#E53E3E; color:#FFF; padding:5px 10px; width:auto; font-size:12px;">حذف</button></td>

</tr>

`;

sellSelect.innerHTML += `<option value="${i.id}">${i.id} - ${i.name} (${i.weight} جم)</option>`;

});

}

 

// Vault Table

const vaultBody = document.getElementById('vaultTableBody');

vaultBody.innerHTML = '';

state.transactions.forEach(t => {

const color = t.amount >= 0 ? '#48BB78' : '#F56565';

vaultBody.innerHTML += `

<tr>

<td>${t.date}</td>

<td>${t.desc}</td>

<td style="color: ${color}; font-weight: bold;">${Math.round(t.amount).toLocaleString()} ج.م</td>

</tr>

`;

});

}
