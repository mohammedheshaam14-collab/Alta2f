function calculateItemPrice(weight, goldPrice, workmanship) {

return weight < 1 ? (goldPrice * weight) + workmanship : (goldPrice + workmanship) * weight;

}

function calcBuy() {

let grand = 0;

for(let i=1; i<=5; i++) {

const w = parseFloat(document.getElementById(`cb_weight_${i}`).value) || 0;

const karat = document.getElementById(`cb_karat_${i}`).value;

const cp = parseFloat(document.getElementById(`cb_price_${i}`).value) || state.k21Price;

const work = parseFloat(document.getElementById(`cb_work_${i}`).value) || 0;

let gp = karat === '24' ? (cp*24)/21 : (karat === '18' ? (cp*6)/7 : cp);

const tot = calculateItemPrice(w, gp, work);

grand += tot;

document.getElementById(`cb_res_${i}`).innerText = Math.round(tot).toLocaleString('ar-EG') + ' ج.م';

}

document.getElementById('calcBuyGrandTotal').innerText = Math.round(grand).toLocaleString('ar-EG') + ' ج.م';

}

 

function calcSwap() {

let oldTot = 0, newTot = 0;

for(let i=1; i<=3; i++) {

const ow = parseFloat(document.getElementById(`so_w_${i}`).value) || 0;

const op = parseFloat(document.getElementById(`so_p_${i}`).value) || 0;

const od = parseFloat(document.getElementById(`so_d_${i}`).value) || 0;

oldTot += calculateItemPrice(ow, Math.max(0, op - od), 0);

const nw = parseFloat(document.getElementById(`sn_w_${i}`).value) || 0;

const np = parseFloat(document.getElementById(`sn_p_${i}`).value) || 0;

const nk = parseFloat(document.getElementById(`sn_k_${i}`).value) || 0;

newTot += calculateItemPrice(nw, np, nk);

}

document.getElementById('swapOldTotalRes').innerText = Math.round(oldTot).toLocaleString('ar-EG') + ' ج.م';

document.getElementById('swapNewTotalRes').innerText = Math.round(newTot).toLocaleString('ar-EG') + ' ج.م';

const diff = newTot - oldTot;

const lbl = document.getElementById('swapFinalStatusLabel');

const res = document.getElementById('swapFinalResult');

if(diff > 0) {

lbl.innerText = "الزبون سيقوم بدفع (فارق نقدي):"; res.className = "text-2xl font-black text-rose-400"; res.innerText = Math.round(diff).toLocaleString('ar-EG') + ' ج.م (دفع)';

} else {

lbl.innerText = "الزبون سيأخذ (فارق نقدي):"; res.className = "text-2xl font-black text-emerald-400"; res.innerText = Math.round(Math.abs(diff)).toLocaleString('ar-EG') + ' ج.م (قبض)';

}

}

function handleBuyGold(e) {

e.preventDefault();

const karat = document.getElementById('buyKarat').value;

const weight = parseFloat(document.getElementById('buyWeight').value) || 0;

if(imageInput.files && imageInput.files[0]) {

reader.onload = function(uploadEvent) {

saveItemToInventory(itemName, karat, weight, cost, uploadEvent.target.result);

};

reader.readAsDataURL(imageInput.files[0]);

} else {

saveItemToInventory(itemName, karat, weight, cost, null);

}

}

 

function saveItemToInventory(itemName, karat, weight, cost, imageUrl) {

state.inventory.push({

id: 'G-' + Math.floor(1000 + Math.random() * 9000),

itemName, karat, weight, pureGoldWeight: (weight*parseInt(karat))/24,

purchaseCost: cost, image: imageUrl, date: new Date().toLocaleString('ar-EG')

});

state.vaultCash -= cost;

state.transactions.unshift({ id: Date.now(), date: new Date().toLocaleString('ar-EG'), type: 'buy', desc: `شراء ${itemName}`, amount: -cost });

document.getElementById('buyForm').reset(); manualTotalChanged = false; saveState(); renderAllData(); switchTab('inventory');

}

function renderInventoryTable() {

const gridContainer = document.getElementById('inventoryGridContainer');

const sel = document.getElementById('sellItemSelect');

gridContainer.innerHTML = ''; sel.innerHTML = '<option value="">-- اختر قطعة --</option>';

document.getElementById('inventoryCountBadge').innerText = state.inventory.length + ' قطعة';

 

state.inventory.forEach(i => {

const imageContent = i.image ?

`<img src="${i.image}" alt="${i.itemName}">` :

`<div class="h-32 flex flex-col items-center justify-center text-gold/60 bg-dark-900/60 rounded-xl border border-gold/20"><i class="fa-solid fa-gem text-3xl mb-1"></i><span class="text-[10px] text-gray-400">الطائف - بدون صورة</span></div>`;

 

gridContainer.innerHTML += `

<div class="jewelry-box space-y-3">

<div class="flex justify-between items-center text-xs border-b border-gold/30 pb-2">

<span class="text-gold font-bold">${i.id}</span>

<span class="text-white font-black">${i.itemName}</span>

</div>

<div>

${imageContent}

</div>

<div class="grid grid-cols-3 gap-1 text-[11px] text-center bg-dark-900/80 p-2 rounded-lg border border-gray-700">

<div>العيار: <span class="text-gold font-bold">${i.karat}</span></div>

<div>الوزن: <span class="text-white font-bold">${i.weight.toFixed(2)} جم</span></div>

<div>التكلفة: <span class="text-emerald-400 font-bold">${Math.round(i.purchaseCost)} ج.م</span></div>

</div>

<div class="text-left pt-1">

<button onclick="deleteInventoryItem('${i.id}')" class="text-xs text-rose-400 hover:text-rose-300 bg-rose-950/40 px-3 py-1.5 rounded-lg border border-rose-800/50 w-full"><i class="fa-solid fa-trash ml-1"></i> حذف القطعة</button>

</div>

</div>

`;

 

sel.innerHTML += `<option value="${i.id}">${i.id} - ${i.itemName} (${i.weight.toFixed(2)} جم)</option>`;

});

}

 

function deleteInventoryItem(id) {

if(confirm('هل أنت متأكد من حذف هذه القطعة؟')) {

state.inventory = state.inventory.filter(x => x.id !== id);

saveState(); renderAllData();

}

}

 

function renderVaultTransactions() {

const tbody = document.getElementById('vaultTransactionsBody'); tbody.innerHTML = '';

state.transactions.forEach(t => {

const col = t.amount > 0 ? 'text-emerald-400' : 'text-rose-400';

tbody.innerHTML += `<tr class="hover:bg-dark-700/50"><td class="p-3 text-gray-400">${t.date}</td><td class="p-3">${t.type}</td><td class="p-3">${t.desc}</td><td class="p-3 font-black ${col}">${t.amount > 0 ? '+' : ''}${Math.round(t.amount).toLocaleString('ar-EG')} ج.م</td></tr>`;

});

}

 

function renderAllData() { renderDashboard(); renderInventoryTable(); renderVaultTransactions(); renderRentSummary(); }
