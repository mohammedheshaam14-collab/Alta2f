const oldAmt = state.transactions[index].amount;

const diff = newAmt - oldAmt;

state.transactions[index].amount = newAmt;

state.vaultCash += diff;

saveState();

renderDashboard();

}

 

function deleteTx(index) {

if(confirm('حذف هذه الحركة النقدية؟')) {

const amt = state.transactions[index].amount;

state.vaultCash -= amt;

state.transactions.splice(index, 1);

saveState(); renderAllData();

}

}
